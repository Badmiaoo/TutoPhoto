#!/usr/bin/env python3
"""Contrôle d'une fiche de lot : chaque constat des rapports du lot y est cité, aucun identifiant inconnu.
Usage : verif_lot.py <dossier de l'index> <fiche du lot> <rapport> [<rapport> …]"""
import json, re, sys
from pathlib import Path
sortie, fiche, cles = Path(sys.argv[1]), Path(sys.argv[2]), sys.argv[3:]
index = json.loads((sortie / "index-constats.json").read_text(encoding="utf-8"))
texte = fiche.read_text(encoding="utf-8")
cites = set(re.findall(r"\b(?:PHY|FEX|FMA|ARC|CHX|SRC|REF|RST|COD|INT|DOC|LIC)-\d+\b", texte))
tous = {c["id"] for r in index["rapports"] for c in r["constats"]}
du_lot = {c["id"]: c["verdict"] for r in index["rapports"] if r["cle"] in cles for c in r["constats"]}
presents = {r["cle"] for r in index["rapports"]}
erreurs = [f"rapport absent de l'index : {c}" for c in cles if c not in presents]
cle = lambda s: (s[:3], int(s[4:]))
m = set(du_lot) - cites
if m: erreurs.append(f"{len(m)} constat(s) du lot absents de la fiche : " + ", ".join(sorted(m, key=cle)))
m = cites - tous
if m: erreurs.append("identifiants qui n'existent dans aucun rapport : " + ", ".join(sorted(m, key=cle)))
m = [i for i, v in du_lot.items() if v is None]
if m: erreurs.append("constats sans verdict dans les rapports : " + ", ".join(sorted(m, key=cle)))
print(f"Lot : {len(du_lot)} constats, {sum(v == 'confirmé' for v in du_lot.values())} confirmés, {sum(v == 'réfuté' for v in du_lot.values())} réfutés")
for e in erreurs: print("ERREUR :", e)
print("Contrôle de la fiche :", "échec" if erreurs else "réussi")
sys.exit(1 if erreurs else 0)
